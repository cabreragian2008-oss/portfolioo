Java Code - Final Performance Task in CC102

Project Title: Number Guessing Game (Console-Based Game)

Description: The computer picks a secret number from 1 to 100. The player guesses with hints like Too High or Too Low, counts attempts, and shows the score at the end.

Concepts Used: Scanner, Random, Loop, Condition, Counter

// Number Guessing Game - CC102 Final PeTa
// By: Gian Carlo N. Cabrera - BSIT 1A

import java.util.Scanner;
import java.util.Random;

public class NumberGuessingGame {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        Random rand = new Random();
        int secretNumber = rand.nextInt(100) + 1;
        int guess = 0;
        int attempts = 0;
        int maxAttempts = 7;

        System.out.println("=== NUMBER GUESSING GAME ===");
        System.out.println("I picked a secret number from 1 to 100.");
        System.out.println("You have " + maxAttempts + " attempts!");

        while (attempts < maxAttempts) {
            System.out.print("Attempt " + (attempts + 1) + " - Enter your guess: ");
            guess = sc.nextInt();
            attempts++;

            if (guess == secretNumber) {
                System.out.println("CONGRATS! You guessed it in " + attempts + " attempts!");
                break;
            } else if (guess < secretNumber) {
                System.out.println("Too Low! Try a bigger number.");
            } else {
                System.out.println("Too High! Try a smaller number.");
            }
            if (attempts == maxAttempts && guess != secretNumber) {
                System.out.println("GAME OVER! Secret number was: " + secretNumber);
            }
        }
        System.out.println("Thank you for playing!");
        sc.close();
    }
}
